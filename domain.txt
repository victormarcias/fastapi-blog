# Redirect www to non-www
server {
    listen 80;
    listen [::]:80;
    server_name victormarcias.online www.victormarcias.online;
    return 301 $scheme://victormarcias.online$request_uri;
}

# Main server block
server {
    listen 80;
    listen [::]:80;
    server_name victormarcias.online;

    # Temporary: serve static files (will change for FastAPI)
    root /var/www/html;
    index index.html;

    location / {
        try_files $uri $uri/ =404;
    }
}
